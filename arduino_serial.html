#define SPEAKER_PIN 8

String line;

void setup() {
  Serial.begin(9600);
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n') { handle(line); line = ""; }
    else if (c != '\r') line += c;
  }
}

void handle(String s) {
  if (s.startsWith("T")) {            // T주파수,길이
    int comma = s.indexOf(',');
    int f  = s.substring(1, comma).toInt();
    int ms = s.substring(comma + 1).toInt();
    tone(SPEAKER_PIN, f, ms);
    Serial.println("OK " + String(f) + "Hz");
  } else if (s == "S") {              // 멈춤
    noTone(SPEAKER_PIN);
    Serial.println("OK stop");
  } else if (s == "R") {              // A0 읽기
    Serial.println("A0 " + String(analogRead(A0)));
  }
}
