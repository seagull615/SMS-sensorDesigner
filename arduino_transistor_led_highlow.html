2. 회로 (C1815, NPN)

트랜지스터는 글자가 있는 평평한 면을 나를 향하게 하고 다리를 아래로 둔 상태에서 왼쪽부터 E - C - B.

#시작부품끝
①아두이노 5V220Ω 저항LED 긴 다리(+)
②LED 짧은 다리(-)직접 연결트랜지스터 C (가운데)
③트랜지스터 E (왼쪽)직접 연결아두이노 GND
④아두이노 13번 핀1kΩ 저항트랜지스터 B (오른쪽)

전류 길: 5V → 220Ω → LED → C → E → GND. 신호선: 13번 핀 → 1kΩ → B.

3. HIGH / LOW 제어

LOW (꺼짐)

HIGH = 13번 핀 5V → 베이스에 전류 → 트랜지스터 ON → LED 켜짐. LOW = 0V → 트랜지스터 OFF → LED 꺼짐.

4. 보드에 올릴 스케치

#define LED_PIN 13

String line;

void setup() {
  Serial.begin(9600);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n') { handle(line); line = ""; }
    else if (c != '\r') line += c;
  }
}

void handle(String s) {
  if (s == "H") {                     // HIGH
    digitalWrite(LED_PIN, HIGH);
    Serial.println("OK HIGH");
  } else if (s == "L") {              // LOW
    digitalWrite(LED_PIN, LOW);
    Serial.println("OK LOW");
  }
}
